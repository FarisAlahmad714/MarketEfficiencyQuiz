<script>
    let chartCount = {{ progress.chart_count|default(1) }};
    const maxCharts = 5;
    let chartScores = [];
    document.getElementById('chart-symbol').textContent = `Symbol: {{ symbol|default('Unknown') }} ({{ interval|default('Unknown') }})`;


    class DrawingLayer {
        constructor(chartContainer, chart, series, chartData) {
            this.container = chartContainer;
            this.chart = chart;
            this.series = series;
            this.chartData = chartData;
            this.svg = document.createElementNS("http://www.w3.org/2000/svg", "svg");
            this.svg.style.position = "absolute";
            this.svg.style.top = "0";
            this.svg.style.left = "0";
            this.svg.style.width = "100%";
            this.svg.style.height = "100%";
            this.svg.style.pointerEvents = "none";
            this.svg.style.zIndex = "5";
            chartContainer.appendChild(this.svg);

            this.currentTool = null;
            this.isDrawing = false;
            this.currentElement = null;
            this.drawings = [];
            this.chartHeight = chartContainer.clientHeight;
            this.chartWidth = chartContainer.clientWidth;

            this.container.addEventListener('mousedown', this.startDrawing.bind(this));
            this.container.addEventListener('mousemove', this.handleMouseMove.bind(this));
            this.container.addEventListener('mouseup', this.endDrawing.bind(this));

            if (this.chartData.length > 0 && 'symbol' in this.chartData[0]) {
                document.getElementById('chart-symbol').textContent = `Symbol: ${this.chartData[0].symbol}`;
            }
        }

        getChartCoordinates(pixelX, pixelY) {
            try {
                if (!this.chartData || this.chartData.length === 0) throw new Error('Chart data not available');
                const timeScale = this.chart.timeScale();
                const visibleLogicalRange = timeScale.getVisibleLogicalRange();
                if (!visibleLogicalRange) throw new Error('Time scale not visible');
                const logicalX = timeScale.coordinateToLogical(pixelX);
                const timeFraction = (logicalX - visibleLogicalRange.from) / (visibleLogicalRange.to - visibleLogicalRange.from);
                const timeRange = { from: this.chartData[0].time, to: this.chartData[this.chartData.length - 1].time };
                let time = timeRange.from + timeFraction * (timeRange.to - timeRange.from);

                const closestCandle = this.chartData.reduce((closest, data) => {
                    const diff = Math.abs(data.time - time);
                    return diff < Math.abs(closest.time - time) ? data : closest;
                }, this.chartData[0]);
                time = closestCandle.time;

                let price = this.series.coordinateToPrice(pixelY);
                if (price === null) {
                    const prices = this.chartData.map(d => [d.low, d.high]).flat();
                    const minPrice = Math.min(...prices);
                    const maxPrice = Math.max(...prices);
                    const priceFraction = 1 - (pixelY / this.chartHeight);
                    price = minPrice + (priceFraction * (maxPrice - minPrice));
                }
                const finalPrice = [closestCandle.low, closestCandle.high].reduce((closest, p) => {
                    const diff = Math.abs(p - price);
                    return diff < Math.abs(closest - price) ? p : closest;
                }, price);

                console.log('Pixel coordinates:', { x: pixelX, y: pixelY });
                console.log('Chart values:', { time, price: finalPrice });
                return { time, price: finalPrice };
            } catch (error) {
                console.error('Error in getChartCoordinates:', error.message);
                return { time: null, price: null };
            }
        }

        setTool(tool) {
            this.currentTool = tool;
            console.log('Tool selected:', tool);
            if (tool === 'line' || tool === 'pointer') {
                this.svg.style.pointerEvents = "auto";
                this.container.style.cursor = tool === 'line' ? 'crosshair' : 'pointer';
            } else {
                this.svg.style.pointerEvents = "none";
                this.container.style.cursor = 'default';
            }
        }

        handleMouseMove(e) { if (this.isDrawing && this.currentTool === 'line') this.draw(e); }

        startDrawing(e) {
            if (!this.currentTool) return;
            this.isDrawing = true;
            const point = this.getMousePosition(e);
            console.log('Started drawing at:', point);
            if (this.currentTool === 'line') {
                this.currentElement = this.createLine(point.x, point.y, point.x, point.y);
                this.svg.appendChild(this.currentElement);
            } else if (this.currentTool === 'pointer') {
                const circle = this.createPoint(point.x, point.y);
                this.svg.appendChild(circle);
                this.drawings.push(circle);
            }
        }

        draw(e) {
            if (!this.isDrawing || this.currentTool !== 'line') return;
            const point = this.getMousePosition(e);
            this.currentElement.setAttribute('x2', point.x);
            this.currentElement.setAttribute('y2', point.y);
            const end = this.getChartCoordinates(point.x, point.y);
            this.currentElement.dataset.endTime = end.time;
            this.currentElement.dataset.endPrice = end.price;
            console.log('Drawing at:', point);
        }

        endDrawing(e) {
            if (!this.isDrawing) return;
            if (this.currentTool === 'line' && this.currentElement) this.drawings.push(this.currentElement);
            this.isDrawing = false;
            this.currentElement = null;
            console.log('Finished drawing');
        }

        createPoint(x, y) {
            const circle = document.createElementNS("http://www.w3.org/2000/svg", "circle");
            const { time, price } = this.getChartCoordinates(x, y);
            circle.setAttribute('cx', x);
            circle.setAttribute('cy', y);
            circle.setAttribute('r', '5');
            circle.setAttribute('fill', 'blue');
            circle.dataset.time = time;
            circle.dataset.price = price;
            return circle;
        }

        createLine(x1, y1, x2, y2) {
            const line = document.createElementNS("http://www.w3.org/2000/svg", "line");
            const start = this.getChartCoordinates(x1, y1);
            const end = this.getChartCoordinates(x2, y2);
            line.setAttribute('x1', x1);
            line.setAttribute('y1', y1);
            line.setAttribute('x2', x2);
            line.setAttribute('y2', y2);
            line.setAttribute('stroke', 'red');
            line.setAttribute('stroke-width', '2');
            line.dataset.startTime = start.time;
            line.dataset.startPrice = start.price;
            line.dataset.endTime = end.time;
            line.dataset.endPrice = end.price;
            return line;
        }

        getMousePosition(event) {
            const rect = this.container.getBoundingClientRect();
            return { x: event.clientX - rect.left, y: event.clientY - rect.top };
        }

        clearDrawings() {
            while (this.svg.firstChild) this.svg.removeChild(this.svg.firstChild);
            this.drawings = [];
            this.setTool(null);
        }

        undoLastDrawing() { if (this.drawings.length > 0) this.svg.removeChild(this.drawings.pop()); }

        updateChartData(newData) {
            this.chartData = newData;
            this.series.setData(newData);
            this.chart.timeScale().fitContent();
            this.clearDrawings();
        }
    }

    function detectSwingPoints(data, lookback = 3) {
        const swingPoints = { highs: [], lows: [] };
        for (let i = lookback; i < data.length - lookback; i++) {
            const current = data[i];
            const before = data.slice(i - lookback, i).map(c => c.high);
            const after = data.slice(i + 1, i + 1 + lookback).map(c => c.high);
            if (current.high > Math.max(...before) && current.high > Math.max(...after)) {
                swingPoints.highs.push({ time: current.time, price: current.high });
            }
            const beforeLows = data.slice(i - lookback, i).map(c => c.low);
            const afterLows = data.slice(i + 1, i + 1 + lookback).map(c => c.low);
            if (current.low < Math.min(...beforeLows) && current.low < Math.min(...afterLows)) {
                swingPoints.lows.push({ time: current.time, price: current.low });
            }
        }
        console.log('Detected swing points:', swingPoints);
        return swingPoints;
    }

    function overlaySwingPoints(expectedPoints) {
        expectedPoints.highs.forEach(point => {
            if (point.price == null) {
                console.error('Skipping null price for high:', point);

            const { x, y } = getPixelCoordinates(point.time, point.price);
            if (x == null || y == null) {
                console.error('Invalid coordinates for high:', point, { x, y });
                return;
            }
            const circle = document.createElementNS("http://www.w3.org/2000/svg", "circle");
            circle.setAttribute('cx', x);
            circle.setAttribute('cy', y);
            circle.setAttribute('r', '5');
            circle.setAttribute('fill', 'green');
            drawingLayer.svg.appendChild(circle);
        });
        expectedPoints.lows.forEach(point => {

            const { x, y } = getPixelCoordinates(point.time, point.price);

           this.isDrawing = false;

           this.drawings = [];

           this.chartWidth = rect.width;
           this.chartHeight = rect.height;
           
           // Calculate price scale width
           this.priceScaleWidth = 50; // Approximate width of price scale
           this.chartAreaWidth = this.chartWidth - this.priceScaleWidth;

           this.container.addEventListener('mousedown', this.startDrawing.bind(this), { capture: true });
           this.container.addEventListener('mousemove', this.handleMouseMove.bind(this), { capture: true });

       getChartCoordinates(pixelX, pixelY) {
           try {
               if (!this.chartData || this.chartData.length === 0) throw new Error('Chart data not available');
               const timeScale = this.chart.timeScale();
               const visibleLogicalRange = timeScale.getVisibleLogicalRange();
               if (!visibleLogicalRange) throw new Error('Time scale not visible');
               const logicalX = timeScale.coordinateToLogical(pixelX);
               const timeFraction = (logicalX - visibleLogicalRange.from) / (visibleLogicalRange.to - visibleLogicalRange.from);
               const timeRange = { from: this.chartData[0].time, to: this.chartData[this.chartData.length - 1].time };
               let time = timeRange.from + timeFraction * (timeRange.to - timeRange.from);

               const closestCandle = this.chartData.reduce((closest, data) => {
                   const diff = Math.abs(data.time - time);
                   return diff < Math.abs(closest.time - time) ? data : closest;
               }, this.chartData[0]);
               time = closestCandle.time;

               const prices = this.chartData.map(d => [d.low, d.high]).flat();
               const minPrice = Math.min(...prices);
               const maxPrice = Math.max(...prices);
               const priceFraction = 1 - (pixelY / this.chartHeight);
               let price = minPrice + (priceFraction * (maxPrice - minPrice));
               const finalPrice = [closestCandle.low, closestCandle.high].reduce((closest, p) => {
                   const diff = Math.abs(p - price);
                   return diff < Math.abs(closest - price) ? p : closest;
               }, price);

               return { time, price: finalPrice };
           } catch (error) {
               console.error('Error in getChartCoordinates:', error.message);
               return { time: null, price: null };
           }
       }

       setTool(tool) {
           this.currentTool = tool;
           if (tool === 'fib') {

               this.container.style.cursor = 'crosshair';
               // Disable chart interaction when drawing
               this.chart.applyOptions({
                   crosshair: { mode: LightweightCharts.CrosshairMode.Disabled },
                   handleScroll: false,
                   handleScale: false
               });
           } else {
               this.svg.style.pointerEvents = "none";
               this.container.style.cursor = 'default';
               // Enable chart interaction when not drawing
               this.chart.applyOptions({
                   crosshair: { mode: LightweightCharts.CrosshairMode.Normal },
                   handleScroll: { mouseWheel: true, pressedMouseMove: true },
                   handleScale: { axisPressedMouseMove: true, mouseWheel: true, pinch: true }
               });
           }
       }

       handleMouseMove(e) {
           if (!this.isDrawing || this.currentTool !== 'fib') return;
           e.preventDefault();
           e.stopPropagation();
           const point = this.getMousePosition(e);

       }

       startDrawing(e) {
           if (!this.currentTool) return;
           e.preventDefault();
           e.stopPropagation();
           this.isDrawing = true;
           const point = this.getMousePosition(e);
           
           if (this.currentTool === 'fib') {
               const coords = this.getChartCoordinates(point.x, point.y);
               this.currentElement = this.createFibonacci(point.x, point.y, point.x, point.y, coords.time, coords.price);
               this.svg.appendChild(this.currentElement.group);
           }
       }
       drawFibonacci(endPoint) {
    if (!this.isDrawing || this.currentTool !== 'fib' || !this.currentElement) return;
    
    // Update the main line position
    const mainLine = this.currentElement.group.querySelector('.fib-main-line');
    mainLine.setAttribute('x2', endPoint.x);
    mainLine.setAttribute('y2', endPoint.y);
    
    // Update end coordinates
    const endCoords = this.getChartCoordinates(endPoint.x, endPoint.y);
    this.currentElement.group.dataset.endTime = endCoords.time;
    this.currentElement.group.dataset.endPrice = endCoords.price;

    // Update the end point marker position
    const markers = this.currentElement.group.querySelectorAll('.fib-point-marker');
    if (markers.length >= 2) {
        const endMarker = markers[1];
        endMarker.setAttribute('cx', endPoint.x);
        endMarker.setAttribute('cy', endPoint.y);
    }

    // Get start and end prices
    const startPrice = parseFloat(this.currentElement.group.dataset.startPrice);
    const endPrice = endCoords.price;
    
    // Store if this is an uptrend or downtrend
    const isUptrend = startPrice < endPrice;
    this.currentElement.group.dataset.isUptrend = isUptrend;
    
    // Calculate the price difference for Fibonacci levels
    const priceDiff = Math.abs(startPrice - endPrice);
    
    // Keep the original Fibonacci levels
    const levels = [0, 0.618, 0.66, 0.705, 1, 1.272, 1.618];
    
    // Clear previous labels
    while (this.labelLayer.firstChild) {
        this.labelLayer.removeChild(this.labelLayer.firstChild);
    }
    
    // Get the Y coordinate for the starting point to ensure level 1 is positioned exactly there
    const startX = parseFloat(this.currentElement.group.dataset.startX);
    const startY = parseFloat(this.currentElement.group.dataset.startY);
    
    // Update each level line and create corresponding label
    this.currentElement.levels.forEach((line, index) => {
        // Calculate the price for this level - following TradingView's approach
        let levelPrice;
        let levelY;
        
        if (levels[index] === 0) {
            // The 0 level is always at the end point (second click)
            levelPrice = endPrice;
            levelY = endPoint.y; // Use exact cursor position for end point
        } else if (levels[index] === 1) {
            // The 1 level is always at the start point (first click)
            levelPrice = startPrice;
            levelY = startY; // Use exact cursor position for start point
        } else {
            // Other levels are calculated proportionally between 0 and 1
            if (isUptrend) {
                // For uptrend: level 0 at top (end), level 1 at bottom (start)
                levelPrice = endPrice - (priceDiff * levels[index]);
                // Calculate the proportional position between start and end
                levelY = endPoint.y + ((startY - endPoint.y) * levels[index]);
            } else {
                // For downtrend: level 0 at bottom (end), level 1 at top (start)
                levelPrice = endPrice + (priceDiff * levels[index]);
                // Calculate the proportional position between start and end
                levelY = endPoint.y + ((startY - endPoint.y) * levels[index]);
            }
        }
        
        // Only show lines that are within the chart
        if (levelY >= 0 && levelY <= this.chartHeight) {
            line.setAttribute('y1', levelY);
            line.setAttribute('y2', levelY);
            line.setAttribute('x1', 0);
            line.setAttribute('x2', this.chartAreaWidth);
            line.style.display = 'block';
        } else {
            line.style.display = 'none';
        }

        // Only create labels for visible lines
        if (levelY >= 0 && levelY <= this.chartHeight) {
            // Create label for this level - TradingView style
            const label = document.createElementNS("http://www.w3.org/2000/svg", "text");
            label.classList.add('fib-level-label');
            label.setAttribute('x', 10); // Left-aligned
            label.setAttribute('y', levelY);
            
            // Format the label text to match TradingView style
            let levelText = levels[index].toString();
            
            label.textContent = `${levelText} (${levelPrice.toFixed(2)})`;
            this.labelLayer.appendChild(label);
            
            // Also add price label on the right side of chart
            const priceLabel = document.createElementNS("http://www.w3.org/2000/svg", "text");
            priceLabel.classList.add('fib-price-label');
            priceLabel.setAttribute('x', this.chartWidth - 5);
            priceLabel.setAttribute('y', levelY);
            priceLabel.setAttribute('text-anchor', 'end');
            priceLabel.textContent = levelPrice.toFixed(2);
            this.labelLayer.appendChild(priceLabel);
        }
    });
    
    // Update the Fibonacci info panel
    updateFibonacciPanel(this.currentElement);
}


       createFibonacci(x1, y1, x2, y2, startTime, startPrice) {
           const group = document.createElementNS("http://www.w3.org/2000/svg", "g");
           group.dataset.startX = x1;
           group.dataset.startY = y1;
           group.dataset.startTime = startTime;
           group.dataset.startPrice = startPrice;
           group.dataset.endX = x2;
           group.dataset.endY = y2;

           // Create main line - yellow dashed like TradingView
           const mainLine = document.createElementNS("http://www.w3.org/2000/svg", "line");
           mainLine.setAttribute('x1', x1);
           mainLine.setAttribute('y1', y1);
           mainLine.setAttribute('x2', x2);
           mainLine.setAttribute('y2', y2);
           mainLine.setAttribute('class', 'fib-main-line');
           group.appendChild(mainLine);
           
           // Add point markers at exact cursor positions
           const startMarker = document.createElementNS("http://www.w3.org/2000/svg", "circle");
           startMarker.setAttribute('cx', x1);
           startMarker.setAttribute('cy', y1);
           startMarker.setAttribute('r', 5);
           startMarker.classList.add('fib-point-marker');
           group.appendChild(startMarker);
           
           const endMarker = document.createElementNS("http://www.w3.org/2000/svg", "circle");
           endMarker.setAttribute('cx', x2);
           endMarker.setAttribute('cy', y2);
           endMarker.setAttribute('r', 5);
           endMarker.classList.add('fib-point-marker');
           group.appendChild(endMarker);

           // Keep the original levels
           const levels = [0, 0.618, 0.66, 0.705, 1, 1.272, 1.618];
           
           const levelLines = levels.map(level => {
               const line = document.createElementNS("http://www.w3.org/2000/svg", "line");
               // Limit line width to chart area excluding price scale
               line.setAttribute('x1', 0);
               line.setAttribute('x2', this.chartAreaWidth);
               line.setAttribute('stroke-width', '1');
               // Use different class names based on level
               line.classList.add(`fib-line-${level.toString().replace('.', '_')}`);
               group.appendChild(line);
               return line;
           });

           return { group, levels: levelLines };
       }

       endDrawing(e) {
           if (!this.isDrawing) return;
           e.preventDefault();

           this.isDrawing = false;
           
           if (this.currentTool === 'fib' && this.currentElement) {
               this.drawings.push(this.currentElement);

           }
       }

       getMousePosition(event) {
           const rect = this.container.getBoundingClientRect();
           return { x: event.clientX - rect.left, y: event.clientY - rect.top };
       }

       clearDrawings() {
           while (this.svg.firstChild) this.svg.removeChild(this.svg.firstChild);
           while (this.labelLayer.firstChild) this.labelLayer.removeChild(this.labelLayer.firstChild);
           while (this.svg.querySelectorAll('.fib-correct-line, .fib-correct-level').length > 0) {
               this.svg.removeChild(this.svg.querySelector('.fib-correct-line'));
               this.svg.querySelectorAll('.fib-correct-level').forEach(line => this.svg.removeChild(line));
           }
           this.drawings = [];
           this.correctDrawings = [];
           this.setTool(null);
           
           // Hide the Fibonacci info panel when clearing drawings
           document.getElementById('fib-info-panel').style.display = 'none';
       }

       undoLastDrawing() {
           if (this.drawings.length > 0) {
               this.svg.removeChild(this.drawings.pop().group);
               
               // Clear all labels first
               while (this.labelLayer.firstChild) {
                   this.labelLayer.removeChild(this.labelLayer.firstChild);
               }
               
               // Recreate labels for remaining drawings
               if (this.drawings.length > 0) {
                   // Update the Fibonacci info panel with the latest drawing
                   updateFibonacciPanel(this.drawings[this.drawings.length - 1]);
               } else {
                   // Hide the panel if no drawings remain
                   document.getElementById('fib-info-panel').style.display = 'none';
               }
           }
       }

   // Function to update the Fibonacci info panel
   function updateFibonacciPanel(drawing) {
       const panel = document.getElementById('fib-info-panel');
       const startPoint = document.getElementById('fib-start-point');
       const endPoint = document.getElementById('fib-end-point');
       const levelsList = document.getElementById('fib-levels-list');
       
       if (!drawing) {
           panel.style.display = 'none';
           return;
       }
       
       // Display the panel
       panel.style.display = 'block';
       
       // Update start and end points with correct labels
       // First point is labeled as 1, second point is labeled as 0
       startPoint.textContent = `$${parseFloat(drawing.group.dataset.startPrice).toFixed(2)} (1)`;
       endPoint.textContent = `$${parseFloat(drawing.group.dataset.endPrice).toFixed(2)} (0)`;
       
       // Calculate Fibonacci levels
       const startPrice = parseFloat(drawing.group.dataset.startPrice); // First point (1.0)
       const endPrice = parseFloat(drawing.group.dataset.endPrice);     // Second point (0.0)
       const priceDiff = Math.abs(startPrice - endPrice);
       const isUptrend = drawing.group.dataset.isUptrend === 'true';
       const levels = [0, 0.618, 0.66, 0.705, 1, 1.272, 1.618];
       
       // Clear previous levels
       levelsList.innerHTML = '';
       
       // Add each level with correct calculation
       levels.forEach(level => {
           let levelPrice;
           
           // Apply same calculation as in drawFibonacci
           if (level === 0) {
               levelPrice = endPrice; // Second point is always 0
           } else if (level === 1) {
               levelPrice = startPrice; // First point is always 1
           } else {
               if (isUptrend) {
                   levelPrice = endPrice - (priceDiff * level);
               } else {
                   levelPrice = endPrice + (priceDiff * level);
               }
           }
           
           const levelDiv = document.createElement('div');
           levelDiv.className = 'fib-level';
           
           const colorSpan = document.createElement('span');
           colorSpan.className = `level-color level-${level.toString().replace('.', '_')}`;
           
           const ratioSpan = document.createElement('span');
           ratioSpan.className = 'level-ratio';
           ratioSpan.textContent = level.toString();
           
           const priceSpan = document.createElement('span');
           priceSpan.className = 'level-price';
           priceSpan.textContent = `$${levelPrice.toFixed(2)}`;
           
           levelDiv.appendChild(colorSpan);
           levelDiv.appendChild(ratioSpan);
           levelDiv.appendChild(priceSpan);
           
           levelsList.appendChild(levelDiv);
       });
   }

class DrawingLayer {
    constructor(chartContainer) {
        this.container = chartContainer;
        this.svg = document.createElementNS("http://www.w3.org/2000/svg", "svg");
        this.svg.style.position = "absolute";
        this.svg.style.top = "0";
        this.svg.style.left = "0";
        this.svg.style.width = "100%";
        this.svg.style.height = "100%";
        this.svg.style.pointerEvents = "all";
        this.svg.style.zIndex = "10"; // Ensure it's on top
        chartContainer.appendChild(this.svg);

        this.currentTool = null;
        this.isDrawing = false;
        this.currentElement = null;
        this.drawings = [];

        this.container.addEventListener('mousedown', this.startDrawing.bind(this));
        this.container.addEventListener('mousemove', this.draw.bind(this));
        this.container.addEventListener('mouseup', this.endDrawing.bind(this));

        // Use addEventListener for buttons
        document.getElementById('line-tool').addEventListener('click', () => this.setTool('line'));
        document.getElementById('pointer-tool').addEventListener('click', () => this.setTool('pointer'));
        document.getElementById('clear-btn').addEventListener('click', () => this.clearDrawings());
        document.getElementById('undo-btn').addEventListener('click', () => this.undoLastDrawing());

        console.log('DrawingLayer initialized');
    }

    setTool(tool) {
        this.currentTool = tool;
        console.log('Tool selected:', tool);
        this.container.style.cursor = tool === 'line' ? 'crosshair' : 'pointer';
    }

    startDrawing(e) {
        if (!this.currentTool) return;
        
        this.isDrawing = true;
        const point = this.getMousePosition(e);
        console.log('Started drawing at:', point);

        if (this.currentTool === 'line') {
            this.currentElement = this.createLine(point.x, point.y, point.x, point.y);
            this.svg.appendChild(this.currentElement);
        } else if (this.currentTool === 'pointer') {
            const circle = this.createPoint(point.x, point.y);
            this.svg.appendChild(circle);
            this.drawings.push(circle);
        }
    }

    draw(e) {
        if (!this.isDrawing || !this.currentElement || this.currentTool !== 'line') return;

        const point = this.getMousePosition(e);
        this.currentElement.setAttribute('x2', point.x);
        this.currentElement.setAttribute('y2', point.y);
        console.log('Drawing at:', point);
    }

    endDrawing(e) {
        if (!this.isDrawing) return;
        
        if (this.currentTool === 'line' && this.currentElement) {
            this.drawings.push(this.currentElement);
        }

        this.isDrawing = false;
        this.currentElement = null;
        console.log('Finished drawing');
    }

    createLine(x1, y1, x2, y2) {
        const line = document.createElementNS("http://www.w3.org/2000/svg", "line");
        line.setAttribute('x1', x1);
        line.setAttribute('y1', y1);
        line.setAttribute('x2', x2);
        line.setAttribute('y2', y2);
        line.setAttribute('stroke', 'lime'); // Brighter color for visibility
        line.setAttribute('stroke-width', '3');
        return line;
    }

    createPoint(x, y) {
        const circle = document.createElementNS("http://www.w3.org/2000/svg", "circle");
        circle.setAttribute('cx', x);
        circle.setAttribute('cy', y);
        circle.setAttribute('r', '5');
        circle.setAttribute('fill', 'purple'); // Brighter color
        return circle;
    }

    getMousePosition(event) {
        const rect = this.container.getBoundingClientRect();
        return {
            x: event.clientX - rect.left,
            y: event.clientY - rect.top
        };
    }

    clearDrawings() {
        while (this.svg.firstChild) {
            this.svg.removeChild(this.svg.firstChild);
        }
        this.drawings = [];
        console.log('Drawings cleared');
    }

    undoLastDrawing() {
        if (this.drawings.length > 0) {
            const lastDrawing = this.drawings.pop();
            this.svg.removeChild(lastDrawing);
            console.log('Last drawing undone');
        }
    }
}

    # Map intervals to valid CoinGecko 'days' parameters
    if interval == '5m':
        days = 1
    elif interval == '4h':
        days = 7
    elif interval == '1d':
        days = 30
    elif interval == '1w':
        days = 365
    else:
        days = 30  # Fallback

    url = f"https://api.coingecko.com/api/v3/coins/{symbol}/ohlc?vs_currency=usd&days={days}"
    headers = {"x-cg-demo-api-key": "CG-X9rKSiVeFyMS6FPbUCaFw4Lc"}  # Replace with your actual API key
    print(f"Fetching CoinGecko data for {symbol} ({interval}, {limit} candles, {days} days)")
    try:
        response = requests.get(url, headers=headers, timeout=10)
        print(f"CoinGecko API response status: {response.status_code}")
        if response.status_code != 200:
            print(f"CoinGecko API error: {response.status_code} - {response.text}")
            return []
        data = response.json()
        print(f"Received {len(data)} candles from CoinGecko")
        candles_to_take = min(limit, len(data))
        candles = [
            {
                'time': int(candle[0]) // 1000,
                'open': float(candle[1]),
                'high': float(candle[2]),
                'low': float(candle[3]),
                'close': float(candle[4]),
                'symbol': symbol.upper() + 'USD'
            }
            for candle in data[-candles_to_take:]
        ]
        print(f"Fetched {len(candles)} candles for {symbol} ({interval})")
        return candles
    except requests.exceptions.RequestException as e:
        print(f"CoinGecko request failed: {str(e)}")
        return []

def validate_swing_points(drawings, section):
    exam_data = session.get('exam_data', {})
    chart_data = exam_data.get('chart_data', [])
    interval = exam_data.get('interval', '4h')
    if not chart_data or len(chart_data) < 10:
        return {
            'success': False,
            'message': 'Insufficient chart data for validation.',
            'score': 0,
            'feedback': {'correct': [], 'incorrect': [{'advice': 'No chart data available.'}]},
            'expected': {'start': {'price': 0, 'time': 0}, 'end': {'price': 0, 'time': 0}},
            'totalExpectedPoints': 4
        }
    
    def pixel_to_price_time(x, y, chart_width=800, chart_height=600):
        times = [c['time'] for c in chart_data]
        prices = [c['high'] for c in chart_data] + [c['low'] for c in chart_data]
        if not times or not prices:
            return 0, 0
        time_range = times[-1] - times[0]
        price_range = max(prices) - min(prices)
        time = times[0] + (x / chart_width) * time_range
        price = min(prices) + (1 - y / chart_height) * price_range
        return time, price

    price_range = max(c['high'] for c in chart_data) - min(c['low'] for c in chart_data) if chart_data else 1
    tolerance_map = {'5m': 0.005, '4h': 0.015, '1d': 0.025, '1w': 0.035}
    price_tolerance = price_range * tolerance_map.get(interval, 0.02)
    time_increment = {'5m': 300, '4h': 14400, '1d': 86400, '1w': 604800}.get(interval, 14400)
    time_tolerance = time_increment * 3

    if section == 'swing_points':
        swing_points = detect_swing_points(chart_data, lookback=5)
        highs = sorted(swing_points['highs'], key=lambda x: x['price'], reverse=True)[:2]
        lows = sorted(swing_points['lows'], key=lambda x: x['price'])[:2]
        required_points = [(p['time'], p['price']) for p in highs] + [(p['time'], p['price']) for p in lows]
        
        if len(required_points) < 4:
            return {
                'success': False,
                'message': 'Not enough significant swing points detected.',
                'score': 0,
                'feedback': {'correct': [], 'incorrect': [{'advice': f"Detected only {len(highs)} highs and {len(lows)} lows—need 2 of each."}]},
                'expected': {
                    'start': {'price': lows[0]['price'] if lows else 0, 'time': lows[0]['time'] if lows else 0},
                    'end': {'price': highs[0]['price'] if highs else 0, 'time': highs[0]['time'] if highs else 0}
                },
                'totalExpectedPoints': 4
            }
        
        matched = 0
        feedback = {'correct': [], 'incorrect': []}
        used_points = set()
        
        for d in drawings:
            if d['type'] == 'line':
                start_t, start_p = pixel_to_price_time(d['start']['x'], d['start']['y'])
                end_t, end_p = pixel_to_price_time(d['end']['x'], d['end']['y'])
                point_matched = False
                
                for i, (rt, rp) in enumerate(required_points):
                    if i in used_points:
                        continue
                    if (abs(start_t - rt) < time_tolerance and abs(start_p - rp) < price_tolerance) or \
                       (abs(end_t - rt) < time_tolerance and abs(end_p - rp) < price_tolerance):
                        matched += 1
                        point_matched = True
                        used_points.add(i)
                        point_type = 'high' if rp in [h['price'] for h in highs] else 'low'
                        feedback['correct'].append({
                            'type': 'swing_point',
                            'time': rt,
                            'price': rp,
                            'advice': f"Good job! You identified a swing {point_type} at price {rp:.2f}."
                        })
                        break
                
                if not point_matched:
                    feedback['incorrect'].append({
                        'type': 'swing_point',
                        'start_time': start_t,
                        'start_price': start_p,
                        'end_time': end_t,
                        'end_price': end_p,
                        'advice': f"This line (start: {start_p:.2f}, end: {end_p:.2f}) doesn’t match a significant swing point."
                    })
        
        for i, (rt, rp) in enumerate(required_points):
            if i not in used_points:
                point_type = 'high' if rp in [h['price'] for h in highs] else 'low'
                feedback['incorrect'].append({
                    'type': 'missed_point',
                    'time': rt,
                    'price': rp,
                    'advice': f"You missed a swing {point_type} at price {rp:.2f}."
                })
        
        total_expected = 4
        success = matched >= total_expected
        score = min(matched / total_expected, 1.0)
        expected = {
            'start': {'price': lows[0]['price'] if lows else 0, 'time': lows[0]['time'] if lows else 0},
            'end': {'price': highs[0]['price'] if highs else 0, 'time': highs[0]['time'] if highs else 0}
        }
        
        return {
            'success': success,
            'message': 'Swing points identified correctly!' if success else 'Some swing points were missed or incorrect.',
            'score': score,
            'feedback': feedback,
            'totalExpectedPoints': total_expected,
            'expected': expected
        }
    else:
        return {
            'success': False,
            'message': f'Validation for section "{section}" not implemented.',
            'score': 0,
            'feedback': {'correct': [], 'incorrect': [{'advice': 'Section not supported yet.'}]},
            'expected': {'start': {'price': 0, 'time': 0}, 'end': {'price': 0, 'time': 0}},
            'totalExpectedPoints': 4
        }

def validate_fibonacci(drawings, chart_data):
    if not chart_data or len(chart_data) < 10:
        return {
            'success': False,
            'message': 'Insufficient chart data for validation.',
            'score': 0,
            'feedback': {'correct': [], 'incorrect': [{'advice': 'No chart data available.'}]},
            'expected': {'start': {'price': 0, 'time': 0}, 'end': {'price': 0, 'time': 0}},
            'totalExpectedPoints': 1
        }
    
    exam_data = session.get('exam_data', {})
    interval = exam_data.get('interval', '4h')
    swing_points = detect_swing_points(chart_data)
    
    if not swing_points['lows'] or not swing_points['highs']:
        return {
            'success': False,
            'message': 'No valid swing points detected.',
            'score': 0,
            'feedback': {'correct': [], 'incorrect': [{'advice': 'No swing points found.'}]},
            'expected': {'start': {'price': 0, 'time': 0}, 'end': {'price': 0, 'time': 0}},
            'totalExpectedPoints': 1
        }
    
    lowest_low = min(swing_points['lows'], key=lambda x: x['price'])
    highest_high = max(swing_points['highs'], key=lambda x: x['price'])
    expected = {'start': lowest_low, 'end': highest_high}
    
    tolerance = get_dynamic_tolerance(interval, chart_data)
    score = 0
    feedback = {'correct': [], 'incorrect': []}
    
    if drawings and len(drawings) > 0:
        user_fib = drawings[0]
        start_diff = abs(user_fib['start']['price'] - expected['start']['price'])
        end_diff = abs(user_fib['end']['price'] - expected['end']['price'])
        if start_diff <= tolerance and end_diff <= tolerance:
            score = 1
            feedback['correct'].append({
                'type': 'fib',
                'start': user_fib['start']['price'],
                'end': user_fib['end']['price'],
                'startTime': user_fib['start']['time'],
                'endTime': user_fib['end']['time'],
                'advice': f"Correct! Fibonacci from {user_fib['start']['price']:.2f} to {user_fib['end']['price']:.2f}."
            })
        else:
            feedback['incorrect'].append({
                'type': 'fib',
                'start': user_fib['start']['price'],
                'end': user_fib['end']['price'],
                'startTime': user_fib['start']['time'],
                'endTime': user_fib['end']['time'],
                'advice': f"Off target - expected start: {expected['start']['price']:.2f}, end: {expected['end']['price']:.2f}."
            })
    else:
        feedback['incorrect'].append({
            'type': 'fib',
            'advice': 'No Fibonacci drawn—draw from swing low to high.'
        })

    success = score > 0
    message = 'Fibonacci retracement correct!' if success else 'Fibonacci placement incorrect.'
    return {
        'success': success,
        'message': message,
        'score': score,
        'feedback': feedback,
        'expected': expected,
        'totalExpectedPoints': 1
    }

def get_dynamic_tolerance(interval, chart_data):
    if not chart_data:
        return 0.02
    price_range = max(c['high'] for c in chart_data) - min(c['low'] for c in chart_data)
    tolerance_map = {'5m': 0.005, '4h': 0.015, '1d': 0.025, '1w': 0.035}
    return price_range * tolerance_map.get(interval, 0.02)

def detect_swing_points(data, lookback=5):
    swing_points = {'highs': [], 'lows': []}
    for i in range(lookback, len(data) - lookback):
        current = data[i]
        before = [c['high'] for c in data[i - lookback:i]]
        after = [c['high'] for c in data[i + 1:i + 1 + lookback]]
        if current['high'] > max(before) and current['high'] > max(after):
            swing_points['highs'].append({'time': current['time'], 'price': current['high']})
        before_lows = [c['low'] for c in data[i - lookback:i]]
        after_lows = [c['low'] for c in data[i + 1:i + 1 + lookback]]
        if current['low'] < min(before_lows) and current['low'] < min(after_lows):
            swing_points['lows'].append({'time': current['time'], 'price': current['low']})
    return swing_points
